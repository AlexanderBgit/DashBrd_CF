// --- 3. KPI-плитки (Швидкий та коректний розрахунок на основі поточного зрізу) --- // 1. Поточна доступна ліквідність на дату звіту (вхідний залишок Tm1) const curCash = ownCash[iTm1]; const curCred = getMlnValue(nodeCredit, tm1Key) || getMlnValue(nodeCredit, "T0"); const curPost = getMlnValue(nodePost, tm1Key); const curTotLiq = curCash + curCred; // 2. Буфер ліквідності (днів) = Доступна ліквідність / Середньодобові витрати з Tm1 const nMandatory = findSnapshotNode(tree, "Обов’язкові платежі") || findSnapshotNode(tree, "Обов'язкові платежі"); const nProcessing = findSnapshotNode(tree, "Переробка"); const nCommercial = findSnapshotNode(tree, "Комерційний оборотний капітал"); const nAllOut = findSnapshotNode(tree, "Всього витрат (1+2+3+4+5)") || findSnapshotNode(tree, "Всього видатків"); let weeklyExp = Math.abs(getMlnValue(nMandatory, tm1Key)) + Math.abs(getMlnValue(nProcessing, tm1Key)) + Math.abs(getMlnValue(nCommercial, tm1Key)); if (weeklyExp === 0) weeklyExp = Math.abs(getMlnValue(nAllOut, tm1Key)); const dailyExp = weeklyExp > 0 ? (weeklyExp / 7) : 1; const bufferDays = dailyExp > 0 ? Math.round(curTotLiq / dailyExp) : 0; // 3. Дельти (Δ) з урахуванням відхилень dev_fact у зрізі const devKey = (nodeCash && nodeCash.values && nodeCash.values.dev_fact !== undefined) ? "dev_fact" : "dev_forecast"; let dLiq = getMlnValue(nodeCash, devKey) + getMlnValue(nodeCredit, devKey); if (dLiq === 0 && nodeCash && nodeCash.values && nodeCash.values.Tm1_plan !== undefined) { const planTotLiq = getMlnValue(nodeCash, "Tm1_plan") + getMlnValue(nodeCredit, "Tm1_plan"); dLiq = curTotLiq - planTotLiq; } const dCred = getMlnValue(nodeCredit, devKey); const dPost = getMlnValue(nodePost, devKey); const prevBuffer = Math.round((curTotLiq - dLiq) / dailyExp); const dBuffer = bufferDays - prevBuffer; // 4. Допоміжна функція форматування динаміки та стрілок const trend = (delta, unitUk, unitEn) => { const a = Math.abs(delta); const num = a === 0 ? "0" : formatNum(a); const arrow = delta > 0 ? "▲ +" : (delta < 0 ? "▼ -" : "— "); return { isPositive: delta >= 0, text: { uk: delta === 0 ? "— без змін" : `${arrow}${num} ${unitUk}`, en: delta === 0 ? "— no change" : `${arrow}${num} ${unitEn}` } }; }; // 5. Формування KPI плиток const kpis = [ { title: { uk: "Доступна ліквідність", en: "Available liquidity" }, value: curTotLiq, unit: { uk: "млн грн", en: "UAH mln" }, trend: trend(dLiq, "млн грн", "UAH mln"), isGray: false }, { title: { uk: "Буфер ліквідності", en: "Liquidity buffer" }, value: bufferDays, unit: { uk: "дні", en: "days" }, trend: trend(dBuffer, "дні", "days"), isGray: false }, { title: { uk: "Доступні кредитні лінії", en: "Available credit lines" }, value: curCred, unit: { uk: "млн грн", en: "UAH mln" }, trend: trend(dCred, "млн грн", "UAH mln"), isGray: false }, { title: { uk: "Доступний постімпорт", en: "Available post-import" }, value: curPost === 0 ? '-' : curPost, unit: { uk: "млн грн", en: "UAH mln" }, trend: trend(dPost, "млн грн", "UAH mln"), isGray: false } ];
